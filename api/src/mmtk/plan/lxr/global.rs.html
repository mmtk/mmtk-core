102    fn collection_required(&self, space_full: bool, _space: Option<SpaceStats<Self::VM>>) -> bool {
103        // Spaces or heap full
104        if self.base().collection_required(self, space_full) {
105            return true;
106        }
107        // SATB is finished
108        if self.concurrent_work_in_progress() && super::concurrent_marking_packets_drained() {
109            return true;
110        }
111        // Survival limits
112        let total_young_alloc_pages =
113            self.block_allocation.total_young_allocation_in_bytes() >> LOG_BYTES_IN_MBYTE;
114        let predicted_survival_mb: usize =
115            ((total_young_alloc_pages as f64 * super::SURVIVAL_RATIO_PREDICTOR.ratio()) as usize)
116                << LOG_CONSERVATIVE_SURVIVAL_RATIO_MULTIPLER;
117        if predicted_survival_mb >= super::MAX_SURVIVAL_MB {
118            return true;
119        }
120        if !self.immix_space.common().contiguous {
121            let available_to_space = self.get_total_pages() - self.get_used_pages();
122            if predicted_survival_mb >= available_to_space {
123                return true;
124            }
125        }
126        false
127    }

149    fn schedule_collection(&'static self, scheduler: &GCWorkScheduler<VM>) {
150        if !super::LazySweepingJobs::all_finished() {
151            warn!("LXR Lazy Sweeping Not Finished");
152        }
153        let pause = self.select_collection_kind();
154        // Wait for concurrent packets
155        // Mark table zeroing
156        if pause == Pause::InitialMark || pause == Pause::Full {
157            self.schedule_mark_table_zeroing_tasks(Some(pause))
158        }
159        self.zeroing_packets_scheduled
160            .store(false, Ordering::SeqCst);
161        // Set current pause kind
162        self.current_pause.store(Some(pause), Ordering::SeqCst);
163        self.perform_cycle_collection
164            .store(pause != Pause::RefCount, Ordering::SeqCst);
165        // Schedule work
166        match pause {
167            Pause::Full => self.schedule_emergency_full_heap_collection(scheduler),
168            Pause::RefCount => self.schedule_rc_collection(scheduler),
169            Pause::InitialMark => self.schedule_concurrent_marking_initial_pause(scheduler),
170            Pause::FinalMark => self.schedule_concurrent_marking_final_pause(scheduler),
171        }
172        // Analysis routine that is ran. It is generally recommended to take advantage
173        // of the scheduling system we have in place for more performance
174        #[cfg(feature = "analysis")]
175        scheduler.work_buckets[WorkBucketStage::Unconstrained].add(GcHookWork);
176        // Resume mutators
177        if pause == Pause::Full || pause == Pause::FinalMark {
178            #[cfg(feature = "sanity")]
179            scheduler.work_buckets[WorkBucketStage::Final].add(ScheduleSanityGC::<Self>::new(self));
180        }
181    }

187    fn prepare(&mut self, _tls: VMWorkerThread) {
188        let pause = self.current_pause().unwrap();
189        if pause == Pause::FinalMark || pause == Pause::Full {
190            self.common.los.is_end_of_satb_or_full_gc = true;
191            // release nursery memory before mature evacuation, to reduce the chance of to-space overflow.
192            self.immix_space.scheduler().work_buckets[WorkBucketStage::Unconstrained]
193                .add(ReleaseLOSNursery);
194        }
195
196        let starts_mark_cycle = pause == Pause::Full || pause == Pause::InitialMark;
197        // Only do prepare if we start a new mark cycle. Otherwise, let thoe spaces have sticky mark bits.
198        if starts_mark_cycle {
199            // We have tested that the following spaces -- they are used by Julia
200            self.common.immortal.prepare();
201            #[cfg(feature = "vm_space")]
202            self.common.base.vm_space.prepare();
203            // TODO: We haven't tested these spaces. But ideally they should be handled in the same way here.
204            self.common.prepare_nonmoving_space(starts_mark_cycle);
205            #[cfg(feature = "code_space")]
206            self.common.base.code_space.prepare();
207            #[cfg(feature = "code_space")]
208            self.common.base.code_lo_space.prepare();
209            #[cfg(feature = "ro_space")]
210            self.common.base.ro_space.prepare();
211        }
212        // LOS is aware of LXR. Call its prepare unconditionally.
213        self.common.los.prepare(starts_mark_cycle);
214
215        if super::MATURE_EVACUATION && (pause == Pause::FinalMark || pause == Pause::Full) {
216            self.process_mature_evacuation_remset();
217        }
218        if super::MATURE_EVACUATION && (pause == Pause::InitialMark || pause == Pause::Full) {
219            // Select mature evacuation set
220            self.schedule_defrag_selection_packets();
221        }
222        self.num_clean_blocks_released_lazy
223            .store(0, Ordering::SeqCst);
224        self.immix_space.prepare_rc(pause);
225        self.block_allocation
226            .reset_block_mark_for_mutator_reused_blocks(pause);
227    }

229    fn release(&mut self, tls: VMWorkerThread) {
230        let _new_ratio = super::SURVIVAL_RATIO_PREDICTOR.update_ratio();
231        let pause = self.current_pause().unwrap();
232        if pause == Pause::FinalMark || pause == Pause::Full {
233            VM::VMCollection::update_weak_processor(false);
234        }
235        <VM as VMBinding>::VMCollection::vm_release();
236        self.common.los.is_end_of_satb_or_full_gc = false;
237        self.common
238            .release(tls, pause == Pause::Full || pause == Pause::FinalMark);
239        self.block_allocation
240            .sweep_nursery_blocks(self.immix_space.scheduler(), pause);
241        self.block_allocation.sweep_mutator_reused_blocks(pause);
242        // Check if we want to do all decs and sweeping in the pause
243        if super::disable_lasy_dec_for_current_gc() {
244            self.immix_space
245                .scheduler()
246                .process_concurrent_packets_in_pause();
247        } else {
248            debug_assert_ne!(pause, Pause::Full);
249        }
250        self.immix_space.release_rc();
251        self.schedule_mature_sweeping(pause);
252        // swap roots
253        let mut prev_roots = self.prev_roots.write().unwrap();
254        let mut curr_roots = self.curr_roots.write().unwrap();
255        std::mem::swap::<SegQueue<_>>(&mut prev_roots, &mut curr_roots);
256        debug_assert!(curr_roots.is_empty());
257    }

290    fn on_pause_start(&self, mmtk: &'static MMTK<Self::VM>) {
291        super::NO_EVAC.store(false, Ordering::SeqCst);
292        let pause = self.current_pause().unwrap();
293
294        // Individual RC pauses that don't overlap with concurrent tracing consist of a GC cycle.
295        // Concurrent tracing, including RC pauses in between, counts as one GC cycle.
296        // A Full GC counts as a GC cycle.
297        if pause == Pause::RefCount && !self.concurrent_work_in_progress()
298            || pause == Pause::InitialMark
299            || pause == Pause::Full
300        {
301            mmtk.gc_trigger.policy.on_gc_start(mmtk);
302        }
303
304        super::SURVIVAL_RATIO_PREDICTOR
305            .set_alloc_size(self.block_allocation.total_young_allocation_in_bytes());
306
307        if pause == Pause::Full || pause == Pause::InitialMark {
308            // Reset block mark and object mark table.
309            let work_packets = self.generate_full_trace_prepare_tasks();
310            self.immix_space.scheduler().work_buckets[WorkBucketStage::RCProcessIncs]
311                .bulk_add(work_packets);
312        }
313
314        for mutator in <VM as VMBinding>::VMActivePlan::mutators() {
315            mutator.flush();
316        }
317
318        if pause == Pause::FinalMark {
319            self.set_concurrent_marking_state(false);
320        }
321    }
322
323    fn on_pause_end(&mut self, mmtk: &'static MMTK<Self::VM>, tls: VMWorkerThread) {
324        super::DISABLE_LASY_DEC_FOR_CURRENT_GC.store(false, Ordering::SeqCst);
325        // self.immix_space.flush_page_resource();
326        let pause = self.current_pause().unwrap();
327        if pause == Pause::InitialMark {
328            self.set_concurrent_marking_state(true);
329        }
330        self.previous_pause.store(Some(pause), Ordering::SeqCst);
331        self.current_pause.store(None, Ordering::SeqCst);
332        LAZY_SWEEPING_JOBS.write().swap();
333        if super::LAZY_DECREMENTS {
334            let perform_cycle_collection =
335                self.get_available_pages() < super::CYCLE_TRIGGER_THRESHOLD;
336            self.hint_cycle_gc
337                .store(perform_cycle_collection, Ordering::SeqCst);
338            self.hint_emergency_gc.store(false, Ordering::SeqCst);
339            self.perform_cycle_collection.store(false, Ordering::SeqCst);
340        }
341        self.avail_pages_at_end_of_last_gc
342            .store(self.get_available_pages(), Ordering::SeqCst);
343        HEAP_AFTER_GC.store(self.get_reserved_pages(), Ordering::SeqCst);
344
345        self.common_mut().on_pause_end(tls);
346
347        // Individual RC pauses that don't overlap with concurrent tracing consist of a GC cycle.
348        // Concurrent tracing, including RC pauses in between, counts as one GC cycle.
349        // A Full GC counts as a GC cycle.
350        if pause == Pause::RefCount && !self.concurrent_work_in_progress()
351            || pause == Pause::FinalMark
352            || pause == Pause::Full
353        {
354            mmtk.gc_trigger.policy.on_gc_end(mmtk);
355        }
356    }

462    fn schedule_mature_sweeping(&self, pause: Pause) {
463        if pause == Pause::Full || pause == Pause::FinalMark {
464            self.evac_set
465                .sweep_mature_evac_candidates(&self.immix_space);
466            let disable_lasy_dec_for_current_gc =
467                crate::plan::lxr::disable_lasy_dec_for_current_gc();
468            let dead_cycle_sweep_packets = self.generate_dead_cycle_sweep_tasks();
469            let sweep_los = RCSweepMatureAfterSATBLOS::new(LazySweepingJobsCounter::new_decs());
470            if super::LAZY_DECREMENTS && !disable_lasy_dec_for_current_gc {
471                debug_assert_ne!(pause, Pause::Full);
472                let concurrent_bucket =
473                    &self.immix_space.scheduler().work_buckets[WorkBucketStage::Concurrent];
474                concurrent_bucket.bulk_add_deferred(dead_cycle_sweep_packets);
475                concurrent_bucket.add_deferred(Box::new(sweep_los));
476            } else {
477                self.immix_space.scheduler().work_buckets[WorkBucketStage::STWRCDecsAndSweep]
478                    .bulk_add(dead_cycle_sweep_packets);
479                self.immix_space.scheduler().work_buckets[WorkBucketStage::STWRCDecsAndSweep]
480                    .add(sweep_los);
481            }
482        }
483    }

494    fn schedule_rc_block_sweeping_tasks(&self, counter: LazySweepingJobsCounter) {
495        // while let Some(x) = self.last_mutator_recycled_blocks.pop() {
496        //     x.set_state(BlockState::Marked);
497        // }
498        // This may happen either within a pause, or in concurrent.
499        let size = self.possibly_dead_mature_blocks.len();
500        let num_bins = self.immix_space.scheduler().num_workers();
501        let bin_cap = size / num_bins + if size % num_bins == 0 { 0 } else { 1 };
502        let mut bins = (0..num_bins)
503            .map(|_| Vec::with_capacity(bin_cap))
504            .collect::<Vec<Vec<(Block, bool)>>>();
505        'out: for bin in bins.iter_mut() {
506            for _ in 0..bin_cap {
507                if let Some(block) = self.possibly_dead_mature_blocks.pop() {
508                    bin.push(block);
509                } else {
510                    break 'out;
511                }
512            }
513        }
514        let packets = bins
515            .into_iter()
516            .map::<Box<dyn GCWork<VM>>, _>(|blocks| {
517                Box::new(SweepBlocksAfterDecs::new(blocks, counter.clone()))
518            })
519            .collect();
520        self.immix_space.scheduler().work_buckets[WorkBucketStage::Unconstrained].bulk_add(packets);
521    }

547    fn next_gc_is_cycle_gc(&self, mature_space_pages: usize, pause: Pause) -> bool {
548        if pause == Pause::FinalMark || pause == Pause::Full {
549            super::MATURE_LIVE_PREDICTOR.update(mature_space_pages);
550        }
551        let live_mature_pages = super::MATURE_LIVE_PREDICTOR.live_pages() as usize;
552        let garbage = mature_space_pages.saturating_sub(live_mature_pages);
553        let total_pages = self.get_total_pages();
554        !self.concurrent_work_in_progress()
555            && (self.cm_enabled() && garbage * 100 >= super::TRACE_THRESHOLD * total_pages)
556    }

558    fn decide_next_gc_may_perform_cycle_collection(&self, pause: Pause) {
559        let (lock, cvar) = &self.decide_cycle_collection;
560        let notify = || {
561            let mut decide_cycle_collection = lock.lock().unwrap();
562            *decide_cycle_collection = true;
563            cvar.notify_one();
564        };
565        // Reset states
566        self.hint_cycle_gc.store(false, Ordering::SeqCst);
567        self.hint_emergency_gc.store(false, Ordering::SeqCst);
568        let emergency_threshold = super::RC_STOP_PERCENT;
569        // Calculate mature space size
570        let total_pages = self.get_total_pages();
571        let mature_space_pages = {
572            let released_los_pages = self.los().num_pages_released_lazy.load(Ordering::SeqCst);
573            HEAP_AFTER_GC
574                .load(Ordering::SeqCst)
575                .saturating_sub(
576                    self.num_clean_blocks_released_lazy.load(Ordering::SeqCst) << Block::LOG_PAGES,
577                )
578                .saturating_sub(released_los_pages)
579        };
580        // Decide next GC kind
581        let hint_cycle_gc = self.next_gc_is_cycle_gc(mature_space_pages, pause);
582        let hint_emergency_gc =
583            self.next_gc_is_emergency_gc(total_pages, mature_space_pages, emergency_threshold);
584        // Update states
585        self.hint_cycle_gc.store(hint_cycle_gc, Ordering::SeqCst);
586        self.hint_emergency_gc
587            .store(hint_emergency_gc, Ordering::SeqCst);
588        // Eager mark-table zeroing
589        if !cfg!(feature = "sanity") && hint_cycle_gc {
590            self.schedule_mark_table_zeroing_tasks(None);
591        }
592        notify();
593    }
594
595    fn schedule_mark_table_zeroing_tasks(&self, pause: Option<Pause>) {
596        if let Some(pause) = pause {
597            assert!(pause == Pause::InitialMark || pause == Pause::Full);
598            if self.zeroing_packets_scheduled.load(Ordering::SeqCst) {
599                return;
600            }
601        }
602        let work_packets = self
603            .immix_space
604            .chunk_map
605            .generate_tasks_batched(self.immix_space.scheduler().num_workers(), |chunks| {
606                Box::new(ConcurrentChunkMetadataZeroing { chunks })
607            });
608        self.immix_space.scheduler().work_buckets[WorkBucketStage::Unconstrained]
609            .bulk_add(work_packets);
610        self.zeroing_packets_scheduled.store(true, Ordering::SeqCst);
611    }

613    fn wait_for_decide_cycle_collection(&self) {
614        let (lock, cvar) = &self.decide_cycle_collection;
615        let mut decide_cycle_collection = lock.lock().unwrap();
616        while !*decide_cycle_collection {
617            decide_cycle_collection = cvar.wait(decide_cycle_collection).unwrap();
618        }
619        *decide_cycle_collection = false;
620    }
621
622    fn select_collection_kind(&self) -> Pause {
623        self.wait_for_decide_cycle_collection();
624
625        let emergency = self.base().global_state.is_emergency_collection();
626        let user_triggered = self.base().global_state.is_user_triggered_collection();
627        let cm_in_progress = self.concurrent_work_in_progress();
628        let cm_packets_drained = super::concurrent_marking_packets_drained();
629        let hint_cycle_gc = self.hint_cycle_gc.load(Ordering::SeqCst);
630        let hint_emergency_gc = self.hint_emergency_gc.load(Ordering::SeqCst);
631        // If CM is finished, do a final mark pause
632        if cm_in_progress && cm_packets_drained {
633            return Pause::FinalMark;
634        }
635
636        // Either final mark pause or full pause for emergency GC
637        if emergency || user_triggered || hint_emergency_gc {
638            return if cm_in_progress {
639                Pause::FinalMark
640            } else {
641                Pause::Full
642            };
643        }
644
645        // Should trigger CM?
646        if hint_cycle_gc && !cm_in_progress {
647            if self.cm_enabled() {
648                Pause::InitialMark
649            } else {
650                Pause::Full
651            }
652        } else {
653            Pause::RefCount
654        }
655    }

657    fn disable_unnecessary_buckets(&'static self, scheduler: &GCWorkScheduler<VM>, pause: Pause) {
658        // Set conditional buckets
659        scheduler.work_buckets[WorkBucketStage::RCProcessIncs].set_enabled(true);
660        scheduler.work_buckets[WorkBucketStage::Prepare].set_enabled(pause != Pause::RefCount);
661        let final_mark_or_full = pause == Pause::FinalMark || pause == Pause::Full;
662        scheduler.work_buckets[WorkBucketStage::Closure].set_enabled(final_mark_or_full);
663        scheduler.work_buckets[WorkBucketStage::WeakRefClosure].set_enabled(final_mark_or_full);
664        scheduler.work_buckets[WorkBucketStage::FinalRefClosure].set_enabled(final_mark_or_full);
665        scheduler.work_buckets[WorkBucketStage::PhantomRefClosure].set_enabled(final_mark_or_full);
666        scheduler.work_buckets[WorkBucketStage::STWRCDecsAndSweep]
667            .set_enabled(!(super::LAZY_DECREMENTS && pause != Pause::Full));
668        // Always enabled
669        scheduler.work_buckets[WorkBucketStage::Concurrent].set_enabled(true);
670        scheduler.work_buckets[WorkBucketStage::ConcurrentResumable].set_enabled(true);
671        // Always disabled
672        scheduler.work_buckets[WorkBucketStage::TPinningClosure].set_enabled(false);
673        scheduler.work_buckets[WorkBucketStage::PinningRootsTrace].set_enabled(false);
674        scheduler.work_buckets[WorkBucketStage::VMRefClosure].set_enabled(false);
675        scheduler.work_buckets[WorkBucketStage::VMRefForwarding].set_enabled(false);
676        scheduler.work_buckets[WorkBucketStage::SoftRefClosure].set_enabled(false);
677        scheduler.work_buckets[WorkBucketStage::CalculateForwarding].set_enabled(false);
678        scheduler.work_buckets[WorkBucketStage::SecondRoots].set_enabled(false);
679        scheduler.work_buckets[WorkBucketStage::RefForwarding].set_enabled(false);
680        scheduler.work_buckets[WorkBucketStage::FinalizableForwarding].set_enabled(false);
681        scheduler.work_buckets[WorkBucketStage::Compact].set_enabled(false);
682    }

684    fn schedule_rc_collection(&'static self, scheduler: &GCWorkScheduler<VM>) {
685        log::info!("Scheduling RC collection...");
686        self.disable_unnecessary_buckets(scheduler, Pause::RefCount);
687        // Before start yielding, wrap all the roots from the previous GC with work-packets.
688        self.process_prev_roots(scheduler);
689        // Stop & scan mutators (mutator scanning can happen before STW)
690        scheduler.work_buckets[WorkBucketStage::Unconstrained]
691            .add(StopMutators::<LXRGCWorkContext<VM>>::new_with_flush());
692        // Prepare global/collectors/mutators
693        scheduler.work_buckets[WorkBucketStage::RCProcessIncs].add(FastRCPrepare);
694        // Release global/collectors/mutators
695        scheduler.work_buckets[WorkBucketStage::Release]
696            .add(Release::<LXRGCWorkContext<VM>>::new(self));
697    }

724    fn schedule_emergency_full_heap_collection(&'static self, scheduler: &GCWorkScheduler<VM>) {

741    fn process_prev_roots(&self, scheduler: &GCWorkScheduler<VM>) {
742        let prev_roots = self.prev_roots.read().unwrap();
743        let mut work_packets: Vec<Box<dyn GCWork<VM>>> = Vec::with_capacity(prev_roots.len());
744        while let Some(decs) = prev_roots.pop() {
745            work_packets.push(Box::new(ProcessDecs::new(
746                decs,
747                LazySweepingJobsCounter::new_decs(),
748            )))
749        }
750        if work_packets.is_empty() {
751            work_packets.push(Box::new(ProcessDecs::new(
752                vec![],
753                LazySweepingJobsCounter::new_decs(),
754            )));
755        }
756        if super::LAZY_DECREMENTS {
757            scheduler.work_buckets[WorkBucketStage::Concurrent].bulk_add_deferred(work_packets);
758        } else {
759            scheduler.work_buckets[WorkBucketStage::STWRCDecsAndSweep].bulk_add(work_packets);
760        }
761    }

779    pub fn mark(&self, o: ObjectReference) -> bool {
780        if self.immix_space.in_space(o) {
781            self.immix_space.attempt_mark(o)
782        } else if self.common.los.in_space(o) {
783            self.common.los.attempt_mark(o)
784        } else {
785            // TODO: We need to properly handle this case.
786            // This is a temporary solution for Julia -- the only other spaces it uses are immortal space and vm space, where objects won't die.
787            debug_assert!(o.is_live());
788            false
789        }
790    }

813    fn on_lazy_sweeping_finished(&self) {
814        self.immix_space.flush_page_resource();
815        // Update counters
816        if !super::LAZY_DECREMENTS {
817            HEAP_AFTER_GC.store(self.get_used_pages(), Ordering::SeqCst);
818        }
819        let pause = match self.current_pause() {
820            Some(p) => p,
821            None => self.previous_pause().unwrap(),
822        };
823        self.decide_next_gc_may_perform_cycle_collection(pause);
824    }

826    fn gc_init(&mut self) {
827        self.immix_space.rc_enabled = true;
828        self.common.los.rc_enabled = true;
829        unsafe {
830            let me: &'static Self = &*(self as *const Self);
831            me.block_allocation.init(&me.immix_space, me);
832            me.immix_space.install_hooks(&me.block_allocation);
833        }
834        let mut lazy_sweeping_jobs = LAZY_SWEEPING_JOBS.write();
835        lazy_sweeping_jobs.swap();
836        let lxr_ptr = self as *const Self as usize;
837        lazy_sweeping_jobs.end_of_decs = Some(Box::new(move |c| {
838            let lxr = unsafe { &*(lxr_ptr as *const Self) };
839            lxr.on_lazy_decs_finished(c);
840        }));
841        lazy_sweeping_jobs.end_of_lazy = Some(Box::new(move || {
842            let lxr = unsafe { &*(lxr_ptr as *const Self) };
843            lxr.on_lazy_sweeping_finished();
844        }));
845    }
